✨ 競技プログラミングの鉄則: 「半分全列挙 (Meet in the Middle)」

対象の要素数が 4 つ（$A, B, C, D$）あってそのままでは指数・高次多項式で爆発するとき、「前半 2 つ」と「後半 2 つ」に分割します。
前半の和 $p = A_i + B_j$ を全列挙したとき、$A_i + B_j + C_k + D_l = K$ が成り立つための条件は、
C_k + D_l = K - p
となります。後半の全通り $C_k + D_l$ を配列 $Q$ にあらかじめ集めてソートしておけば、値 $(K - p)$ が存在するかどうかを $O(\log N)$ で高速判定できます。

模範解答コード (A14)

C++ の `std::binary_search` または Python の `bisect` を使用した鉄則解法です。

#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    long long K;
    if (!(cin >> N >> K)) return 0;

    vector<long long> A(N), B(N), C(N), D(N);
    for (int i = 0; i < N; i++) cin >> A[i];
    for (int i = 0; i < N; i++) cin >> B[i];
    for (int i = 0; i < N; i++) cin >> C[i];
    for (int i = 0; i < N; i++) cin >> D[i];

    // 1. 前半グループ (A + B) の和をすべて列挙 (N^2 通り)
    vector<long long> P;
    P.reserve(N * N);
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            P.push_back(A[i] + B[j]);
        }
    }

    // 2. 後半グループ (C + D) の和をすべて列挙 (N^2 通り)
    vector<long long> Q;
    Q.reserve(N * N);
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            Q.push_back(C[i] + D[j]);
        }
    }

    // 3. 後半配列 Q を昇順にソート (二分探索のため)
    sort(Q.begin(), Q.end());

    // 4. P の各要素に対し、相方 (K - p) が Q に存在するか二分探索
    for (long long p : P) {
        long long target = K - p;
        if (binary_search(Q.begin(), Q.end(), target)) {
            cout << "Yes\n";
            return 0; // 見つかったら即座に終了
        }
    }

    cout << "No\n";
    return 0;
}
import sys
import bisect

def main():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return

    N = int(data[0])
    K = int(data[1])

    idx = 2
    A = [int(x) for x in data[idx : idx + N]]; idx += N
    B = [int(x) for x in data[idx : idx + N]]; idx += N
    C = [int(x) for x in data[idx : idx + N]]; idx += N
    D = [int(x) for x in data[idx : idx + N]]; idx += N

    # 1. 前半 (A + B) の和を列挙 (N^2 通り)
    P = [a + b for a in A for b in B]

    # 2. 後半 (C + D) の和を列挙 (N^2 通り)
    Q = [c + d for c in C for d in D]

    # 3. 後半配列 Q をソート
    Q.sort()

    # 4. P の各要素に対し、K - p を二分探索 (bisect)
    len_q = len(Q)
    for p in P:
        target = K - p
        pos = bisect.bisect_left(Q, target)
        if pos < len_q and Q[pos] == target:
            print("Yes")
            return

    print("No")

if __name__ == "__main__":
    main()
